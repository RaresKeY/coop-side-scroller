cmake_minimum_required(VERSION 3.20)
project(EmberCore VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(ember_core SHARED core/src/ember_core.cpp)
add_library(ember_core_static STATIC core/src/ember_core.cpp)
set_target_properties(ember_core_static PROPERTIES OUTPUT_NAME ember_core_static)
foreach(target ember_core ember_core_static)
 target_include_directories(${target} PUBLIC core/include)
 set_target_properties(${target} PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES POSITION_INDEPENDENT_CODE ON)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /GR- /EHs-c-)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)
 endif()
endforeach()
if(UNIX AND NOT APPLE)
 target_link_options(ember_core PRIVATE "LINKER:--as-needed")
endif()
target_compile_definitions(ember_core PRIVATE EMBER_BUILD)
target_compile_definitions(ember_core_static PUBLIC EC_STATIC)
add_executable(ember_standalone examples/standalone.cpp)
target_link_libraries(ember_standalone PRIVATE ember_core_static)
include(CTest)
if(BUILD_TESTING)
 add_executable(ember_contract_tests core/tests/core_tests.cpp)
 target_link_libraries(ember_contract_tests PRIVATE ember_core_static)
 add_executable(ember_c_consumer core/tests/c_consumer.c)
 target_link_libraries(ember_c_consumer PRIVATE ember_core)
 foreach(test_target ember_contract_tests ember_c_consumer)
  if(MSVC)
   target_compile_options(${test_target} PRIVATE /UNDEBUG)
  else()
   target_compile_options(${test_target} PRIVATE -UNDEBUG)
  endif()
 endforeach()
 add_test(NAME core_contract COMMAND ember_contract_tests)
 add_test(NAME c_abi_consumer COMMAND ember_c_consumer)
 add_test(NAME standalone_replication COMMAND ember_standalone)
endif()
